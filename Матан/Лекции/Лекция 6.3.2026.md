### Теорема 3:
$f(x)$ монотонна на $[a; b] \implies f(x) \in \mathbb{R}[a; b]$
Доказательство:
$$
\begin{align*}
\forall{\epsilon > 0}: \forall{p}: \sum_{j = 1}^{n_{p}}{\omega(f, \Delta_{j})\Delta x_{j}} = \sum_{j = 1}^{n_{p}}{|f(x_{j})-f(x_{j-1})|\underbrace {\Delta x_{j}}}_{\leq \lambda} \leq \\
\leq  \lambda_{p} \sum_{j = 1}^{n_{p}}{|f(x_{j}) - f(x_{j-1})|} = \\
= \lambda_{p}|\sum_{j=1}^{n_{p}}{f(x_{j}) - f(x_{j-1})}| = \lambda_{p}|f(b) - f(a)| < \epsilon \implies \lambda_{p} < \frac{\epsilon}{|f(b) - f(a)|}
\end{align*}
$$
### Определение:
$l \in (-\infty; + \infty)$ имеет меру О по Лебегу, если $\forall{\epsilon>0} \exists{\{ (a_{j}; b_{j}) \}}:$
1. $l \subset U_{j}(a_{j}; b)_{j}$
2. $\sum{\text{длина }(a_{j}; b_{j})} < \epsilon$
### Критерий Лебега
$f(x) \in \mathbb{B}[a; b], f(x) \in \mathbb{R}[a; b] \Longleftrightarrow$ множество её точек разрыва меры О по Лебегу. 

## Свойства определённого интеграла
### Теорема 1:
Функция $f(x)$ и $g(x)$ интегрируемы на $[a; b] \implies \forall{\alpha, \beta \in \mathbb{R} (\mathbb{C})} \implies \alpha f + \beta g \in \mathbb{R}[a; b]$ и $\int_{a}^{b}{(\alpha f + \beta g)dx} = \alpha\int_{a}^{b}{f(x)dx} + \beta\int_{a}^{b}{g(x)dx}$
#### Доказательство:
$$
\begin{align*}
\forall{p}, \forall{\xi}: \sigma(\alpha f + \beta g, p, \xi) = \sum_{j = 1}^{n_{p}}{(\alpha f(\xi_{j}) + \beta g(\xi_{j}))\Delta x_{j}}  \\
= \alpha \sum_{j = 1}^{n_{p}}{f(\xi_{j})\Delta x_{j}} +\beta \sum_{j=1}^{n_{p}}{g(\xi_{j})\Delta x_{j}} \\
\lambda_{p} \to 0 \implies \alpha \sum_{j = 1}^{n_{p}}{f(\xi_{j})\Delta x_{j}} \to \alpha\int_{a}^{b}{f(x)dx} \quad \beta \sum_{j=1}^{n_{p}}{g(\xi_{j})\Delta x_{j}} \to \beta\int_{a}^{b}{g(x)dx} \\
\square
\end{align*}
$$

### Теорема 2
$f(x) \in \mathbb{R}[a; b] \implies |f(x)| \in \mathbb{R}[a; b]$
#### Доказательство (через неравенство треугольника):
$$
\begin{align*}
|c - d| \geq | |c| - |d| | \implies  \forall{x', x''}~ |f(x') - f(x'')| \geq  ||f(x')| - |f(x'')|| \implies \\
\implies \forall{\Delta} ~ \omega(f, \Delta) \geq \omega(|f|, \Delta) \implies  \\
\text{По критерию д'Арбу} \sum_{j = 1}^{n_{p}}{\omega(f, \Delta_{j})\Delta x_{j}} \geq \sum_{j = 1}^{n_{p}}{\omega(|f|, \Delta_{j})\Delta x_{j}} \geq 0 \\
\sum_{j = 1}^{n_{p}}{\omega(f, \Delta_{j})\Delta x_{j}} \to 0 \implies  \sum_{j = 1}^{n_{p}}{\omega(|f|, \Delta_{j})\Delta x_{j}} \to \implies |f| \in \mathbb{R}[a; b] \quad \square
\end{align*}
$$

### Теорема 3
$f(x), g(x) \in \mathbb{R}[a; b] \implies f\cdot g \in \mathbb{R}[a; b]$
#### Доказательство:
1. $f(x) \in \mathbb{R}[a;b] \implies f^{2}\in \mathbb{R}[a; b]$
$$
\begin{align*}
f(x) \in \mathbb{B}[a; b] \implies \exists M>0: \forall{x', x''}: |f(x)| < M \\
|f^{2}(x') - f^{2}(x'')| = |f(x') + f(x'')| \cdot |f(x') - f(x'')| \leq \\
\leq  (|f(x')| + |f(x'')|) \cdot |f(x') - f(x'')|\leq 2M\cdot |f(x') - f(x'')| \\
\forall{p}: \lambda_{p} \to 0: \sum_{j=1}^{n_{p}}{\omega(f^{2}, \Delta_j)\Delta x_{j}} \leq 2M \sum_{j=1}^{n_{p}}{\omega(f, \Delta_{j})\Delta x_{j}} \to 0 \implies f^{2}\in \mathbb{R}[a; b]
\end{align*}
$$
2. $f(x), g(x) \in \mathbb{R}[a; b] \implies f + g \in \mathbb{R}[a; b] \text{ и } f - g \in \mathbb{R}[a; b]$
$$
\begin{align*}
(f+g)^{2} \in \mathbb{R}[a; b] \text{ и } (f - g)^{2} \in \mathbb{R}[a; b] \implies \frac{1}{4}((f+g)^{2} - (f-g)^{2}) =fg \in \mathbb{R}[a; b]
\end{align*}
$$
## Аддитивность интеграла как функции отрезка:
### Утверждение 1:
$f(x) \in \mathbb{R}[a; b] \text{ и } [c; d] \subset [a; b] \implies f(x) \in \mathbb{R}[c; d]$
#### Доказательство:
$$
\begin{align*}
\forall{\epsilon>0} ~ \exists{\delta_{\epsilon}>0}: \forall{p, \lambda_{p}<\delta_{\epsilon}} \sum_{j=1}^{n_p}{\omega(f, \Delta_{j})\Delta x_{j}}<\epsilon \\
\forall{\pi} \text{ Разбиение } [c; d], \lambda_{\pi}<\delta_{\epsilon} \\
\text{Дополним } \pi \text{ до разбиения } [a; b] \text{ добавляяя точки на } [a; b]\[c; d], \\
 \text{ так чтобы }\lambda_{p} \leq  \lambda_{\pi} <\delta \\
\epsilon > \sum_{j=1}^{n_{p}}{\omega(f, \Delta_{j})\Delta x_{j}}\geq \sum_{j: x_{j} \in [c; d]}^{n_{p}}{\omega(f, \Delta_{j})\Delta x_{j}} \implies f(x) \in \mathbb{R}[c; d]  \quad\square
\end{align*}
$$

### Утверждение 2:
$$
\begin{align*}
a  < c < b, f(x) \in \mathbb{R}[a; c], f(x) \in \mathbb{R}[c; d] \implies f(x)\in \mathbb{R}[a; b]  \\
\text{ и } \int_{a}^{b}{f(x)dx} = \int_{a}^{c}{f(x)dx} + \int_{c}^{b}{f(x)dx}
\end{align*}
$$
#### Доказательство (проверка критерия д'Арбу):
1. Ограниченность: $f(x) \in \mathbb{B}[a; c]\cup [c; b] \implies f(x) \in \mathbb{B}[a; b]$
2. 
$$
\begin{align*}
\forall{\epsilon >0} \\
\exists{\delta_{\epsilon}'>0}: \forall_{[a; c]}{p}: \lambda_{p}<\delta_{\epsilon}' \sum_{j = 1}^{n_{p, [a; c]}}{\omega(f, \Delta_{j})\Delta x_{j}}<\epsilon \\
\exists{\delta_{\epsilon}''>0}: \forall_{[c; b]}{p}: \lambda_{p}<\delta_{\epsilon}'' \sum_{j = 1}^{n_{p, [c; b]}}{\omega(f, \Delta_{j})\Delta x_{j}}<\epsilon \\
\delta_{\epsilon} = \min\{ \epsilon, \delta_{\epsilon}', \delta_{\epsilon}'' \} > 0 \\
\forall{p}_{[a; b]} \lambda_{p, [a; b]} < \delta_{\epsilon} ~ \sum_{j =1}^{n_{p, [a; b]}}{\omega(f, \Delta_{j})\Delta x_{j}} = \underbrace{\sum_{j: \Delta_{j} \subset [a; c)}'}_{<\epsilon} + \sum_{j: \Delta_{j} \cap c \neq \emptyset}'' + \underbrace {\sum_{j: \Delta_{j} \subset (c; b]}'''}_{<\epsilon} \\
 < 2\epsilon + 2M \cdot \left( \sum{\text{длина } \Delta_{j}, \text{ где } \Delta_{j} \ni c} \right)\cdot2 = 2\epsilon + 4M\epsilon = \epsilon(2 + 4M) \implies  \\
 \implies  f(x) \in \mathbb{R}[a;b]
\end{align*}
$$
$$
\begin{align*}
\forall{p}_{[a; b]} \ni c, \lambda_{p, [a; b]} \to 0: \forall{\xi} \quad \sigma(f, p_{[a; b]}, \xi) = \sigma(f, p_{[a; c]}, \xi) + \sigma(f, p_{[c; d]}, \xi) \implies  \\
\implies \int_{a}^{b}{f(x)dx} = \int_{a}^{c}{f(x)dx} + \int_{c}^{b}{f(x)dx}
\end{align*}
$$
### Следствие:
$$
f(x) \in \mathbb{R}[a; b] \implies  \forall{c \in [a; b]}: \int_{a}^{b}{f(x)dx} = \int_{a}^{c}{f(x)dx} + \int_{c}^{b}{f(x)dx}
$$
