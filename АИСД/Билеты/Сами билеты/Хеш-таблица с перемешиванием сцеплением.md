Таблица - массив указателей на головы связных списков (цепочек). По факту является открытой таблицей с закрытой адресацией.
## Операции:
### Вставка:
Вычислить индекс ind = h(k). Проверить, нет ли ключа в списке table[ind].
Добавить элемент в начало списка table[ind].
### Поиск:
Вычислить ind = h(k)
Линейный поиск в списке table[ind]

### Удаление:
Найти элемент в списке table[ind] и удалить узел

## Оценка эффективности:
Коэффициент заполнения: $\sigma = \frac{n}{m}$ (средняя длина цепочки)
Средний случай: O(1 + $\sigma$)
Худший случай: O(n) (все ключи попали в 1 цепочку).

Полный код:
```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef char* KeyType; 
typedef unsigned long InfoType;
typedef unsigned long IndexType;

typedef struct Space{
    KeyType key;
    InfoType info;
    struct Space *next;
}Space;

typedef struct Table{
    IndexType size;
    IndexType capacity;
    struct Space **heads;
}Table;

IndexType hash(KeyType key){
    IndexType h=0;
    for (int i=0; key[i]!='\0'; i++){
        h=h*37+(unsigned char)key[i];
    }
    return h;
}

Table* create(IndexType capacity){
    Table* table=(Table*)malloc(sizeof(Table));
    if (table==NULL) return NULL;
    table->size=0;
    table->capacity=capacity;
    table->heads=(Space**)calloc(capacity,sizeof(Space*));
    if (table->heads==NULL) {
        free(table);
        return NULL;
    }
    return table;
}

int compare(KeyType key1,KeyType key2){
    return strcmp(key1,key2);
}

Space* find(Table *table, KeyType key){
    if (table==NULL || key==NULL) return -1;
    IndexType id=hash(key)%table->capacity;
    Space *cur=table->heads[id];
    while (cur){
        if (compare(cur->key, key)==0){
            return cur;
        }
        cur=cur->next;
    }
    return NULL;
}

int insert(Table *table, KeyType key, InfoType info){
    if (table==NULL || key==NULL) return -1;
    if (find(table, key)!=-1) return -1;
    IndexType id=hash(key)%table->capacity;
    Space *new=(Space*)malloc(sizeof(Space));
    if (new==NULL) return -1;
    new->info=info;
    new->key=strdup(key);
    new->next=table->heads[id];
    table->heads[id]=new;
    table->size++;
    return 0;
}

int delete(Table *table, KeyType key, InfoType *output){
    if (table==NULL || key==NULL) return -1;
    IndexType id=hash(key)%table->capacity;
    Space **cur=&table->heads[id];
    while (*cur){
        if (compare((*cur)->key, key)==0){
            Space* time=*cur;
            *output=time->info;
            *cur=time->next;
            free(time->key);
            free(time);
            table->size--;
            return 0;
        }
        cur=&(*cur)->next;
    }
    return -1;   
}



int delete(Table *table, KeyType key, InfoType *output){
    if (table==NULL || key==NULL) return -1;
    IndexType id=hash(key)%table->capacity;
    Space *cur=table->heads[id];
    Space *prev=NULL;
    while (cur){
        if (compare(cur->key, key)==0){
            *output=cur->info;
            if (prev==NULL){
                table->heads[id]=cur->next;
            }else{
                prev->next=cur->next;
            }
            free(cur->key);
            free(cur);
            table->size--;
            return 0;
        }
        prev=cur;
        cur=cur->next;
    }
    return -1;   
}

void kill(Table *table){
    for (IndexType id=0; id<table->capacity; id++){
        Space *cur=table->heads[id];
        while (cur){
            Space* time=cur->next;
            free(cur->key);
            free(cur);
            cur=time;
        }
    }
    free(table->heads);
    free(table);
}
```
