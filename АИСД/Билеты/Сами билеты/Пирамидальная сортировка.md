typedef struct Heap{
    int *data;
    int size;
    int capacity;
}Heap;

Heap* create(int capacity){
    Heap *heap=(Heap*)malloc(sizeof(Heap));
    heap->data=(int*)malloc(sizeof(int)*capacity);
    heap->size=0;
    heap->capacity=capacity;
    return heap;
}

int max_heap(Heap* heap, int i){
    int largest=i;
    int left=2*i+1;
    int right=2*i+2;
    if (left<heap->size && heap->data[left]>heap->data[largest]) largest=left;
    if (right<heap->size && heap->data[right]>heap->data[largest]) largest=right;
    if (largest!=i){
        int time=heap->data[i];
        heap->data[i]=heap->data[largest];
        heap->data[largest]=time;
        max_heap(heap, largest);
    }
    return 0;
}

void build_max_heap(Heap* heap){
    for (int i=heap->size/2-1; i>=0; i--){
        max_heap(heap, i);
    }
}

void heap_sort(Heap *heap) {
    build_max_heap(heap);                // строим max-кучу
    int original_size = heap->size;      // сохраняем исходный размер
    for (int i = heap->size - 1; i > 0; i--) {
        // меняем корень (макс) с i-м элементом (последним в текущей куче)
        int temp = heap->data[0];
        heap->data[0] = heap->data[i];
        heap->data[i] = temp;
        heap->size--;                    // уменьшаем размер кучи
        max_heap(heap, 0);               // восстанавливаем кучу для корня
    }
    heap->size = original_size;          // возвращаем размер (массив теперь отсортирован)
}

**Пример сортировки.** Исходный массив: `[4,10,3,5,1]`, size=5.

1. После `build_max_heap`: `[10,5,3,4,1]` (куча: size=5).
    
2. i=4: меняем 10 и 1 → `[1,5,3,4,10]`, size-- (=4), вызываем `max_heap(heap,0)`.  
    Просеивание: 1 и дети 5,3. 5>1 => обмен 1 и 5 → `[5,1,3,4,10]`, затем для индекса 1 дети 4 (индекс 3) и нет правого? left=3 (4) < size=4 => 4>1, обмен → `[5,4,3,1,10]`.  
    Куча (size=4): `[5,4,3,1]`. Отсортировано: `[10]`.
    
3. i=3: меняем 5 и 1 → `[1,4,3,5,10]`, size-- (=3), `max_heap(0)`. Дети: 4,3. 4>1 => обмен → `[4,1,3,5,10]`. Для индекса 1 дети: left=3 (size=3, нет). Готово: куча `[4,1,3]`.
    
4. i=2: меняем 4 и 3 → `[3,1,4,5,10]`, size=2, `max_heap(0)`. Левый ребёнок 1 (индекс 1, значение 1) < size=2, но 1<3, без обмена. Куча `[3,1]`.
    
5. i=1: меняем 3 и 1 → `[1,3,4,5,10]`, size=1, `max_heap(0)` не делает ничего. Цикл завершён.  
    Восстанавливаем `heap->size = 5`. Массив отсортирован по возрастанию: `[1,3,4,5,10]`.

Идея heapsort:

1. Построить max-heap (максимальный элемент в корне).
    
2. Менять местами корень (максимум) с последним элементом кучи, уменьшать размер кучи на 1 (исключая отсортированный хвост).
    
3. Восстанавливать свойство кучи для нового корня (просеиванием вниз).
    
4. Повторять, пока в куче больше одного элемента.

Время (лучший/средний/худший) -O(n log n)

Память - O(1) (сортировка на месте)
