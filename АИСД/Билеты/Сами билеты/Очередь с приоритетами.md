Структура данных для обслуживания множества S, где каждый элемент имеет ключ (приоритет). Элементы извлекаются в порядке приоритета. По сути та же самая [[Пирамида|пирамида]]

### Основные операции (для max - очереди)
#### Maximum (S)
Вернуть элемент с наибольшим ключом (не удаляя). O(1)
#### Extract_Max(S)
Вернуть и удалить элемент с наибольшим ключом. O($\log{n}$)

### Insert(S, x)
Вставить элемент x. O($\log{n}$)

#### Increase_Key(S, i, k)
Увеличить ключ элемента i до k (k >= current).

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX 10000

typedef struct Priority_Queue{
    int capacity;
    int size;
    int *data;
}Priority_Queue;

Priority_Queue* create(int capacity){
    Priority_Queue* queue=(Priority_Queue *)malloc(sizeof(Priority_Queue)*capacity);
    queue->size=0;
    queue->capacity=capacity;
    queue->data=(int*)malloc(sizeof(int)*capacity);
    return queue;
}

int max_heapify(Priority_Queue *queue, int i){
    int largest=i;
    int left=2*i+1;
    int right=2*i+2;
    if (left<queue->size && queue->data[left]>queue->data[largest]) largest=left;
    if (right<queue->size && queue->data[right]>queue->data[largest]) largest=right;
    if (i!=largest){
        int time=queue->data[i];
        queue->data[i]=queue->data[largest];
        queue->data[largest]=time;
        max_heapify(queue, largest);
    }
    return 0;
}

int build_max_heap(Priority_Queue *queue){
    for (int i=(queue->size-1)/2; i>=0; i--){
        max_heapify(queue, i);
    }
    return 0;
}

int insert(Priority_Queue *queue, int val){
    int i=queue->size++;
    queue->data[i]=val;
    while (i>0){
        int parent = (i-1)/2;
        if (queue->data[parent]<queue->data[i]) {
            int tmp=queue->data[i];
            queue->data[i]=queue->data[parent];
            queue->data[parent]=tmp;
            i=parent;
        }else{
            break;
        }
    }
    return 0;
}

int maximum(Priority_Queue* queue){
    return queue->data[0];
}

int extract_max(Priority_Queue *queue){
    if (queue==NULL) return -1;
    int time=queue->data[0];
    queue->data[0]=queue->data[queue->size-1];
    queue->size=queue->size-1;
    max_heapify(queue,0);
    return time;
}

int increase_key(Priority_Queue *queue, int i, int key){
    if (queue==NULL) return -1;

        queue->data[i]=queue->data[par];
        queue->data[par]=time;
        i=par;
    }
    return 0;
}
