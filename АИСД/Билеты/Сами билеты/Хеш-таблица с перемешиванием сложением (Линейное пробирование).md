Закрытая таблиц с отрытой адресацией.
Проблема: первичная кластеризация: образуются сплошные блоки занятых ячеек. Чем больше блок, тем выше вероятность, что новый элемент попадет в него и увеличит его ещё больше. Это ухудшает производительность. 

Полный код:
```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef char* KeyType; 
typedef unsigned long InfoType;
typedef unsigned long IndexType;

typedef struct Node{
    KeyType key;
    InfoType info;
    IndexType busy;
}Node;

typedef struct Table{
    IndexType size;
    IndexType capacity;
    Node *node;
    IndexType step;
}Table;

int hash(KeyType key){
    int h=0;
    for (int i=0; key[i]!='\0'; i++){
        h=h*37+(unsigned char)key[i];
    }
    return h;
}

int compare(KeyType key1, KeyType key2){
    return strcmp(key1, key2);
}

Table* create(int capacity){
    Table* table=(Table*)malloc(sizeof(Table));
    table->capacity=capacity;
    table->size=0;
    table->node=(Node*)calloc(capacity,sizeof(Node));
    table->step=3;
    return table;
}

int find(Table *table, KeyType key, Node **output){
    if (table==NULL || key==NULL) return -1;
    for (int i=0; i<table->capacity; i++){
        IndexType has=(hash(key)+i*table->step)%table->capacity;
        if (table->node[has].busy==0) break;
        if (compare(table->node[has].key,key)==0 && table->node[has].busy==1){
            *output=&(table->node[has]);
            return 0;
        }
    }
    return -1;
}

int insert(Table *table, KeyType key, InfoType info){
    if (table==NULL || key==NULL || table->size>=table->capacity) return -1;
    if (table->size>=table->capacity) return -1;
    Node *output;
    if (find(table,key, &output)!=NULL) return -1;
    int index=-1;
    int deleted=-1;
    for (int j=0; j<table->capacity; j++){
        IndexType has=(hash(key)+j*table->step)%table->capacity;
        if (table->node[has].busy==0){
            index=has;
            break;
        }
        if (table->node[has].busy==2){
            deleted=has;
        }
    }
    if (index==-1) index=deleted;
    if (index==-1) return -1;
    table->node[index].key=strdup(key);
    table->node[index].info=info;
    table->node[index].busy=1;
    table->size++;
    return 0;
}

int delete(Table *table, KeyType key ,InfoType *output){
    if (table==NULL) return -1;
    Node *out;
    if (find(table, key, &out)==-1) return -1;
    *output=out->info;
    free(out->key);
    out->busy=2;
    out->info=NULL;
    table->size--;
    return 0;
}

int delete(Table *table, KeyType key){
    if (table==NULL) return -1;
    for (int j=0; j<table->capacity; j++){
        IndexType hash_value=(hash(key)+table->step*j)%table->capacity;
        if (table->node[hash_value].busy==0) break;
        if (compare(table->node[hash_value].key,key) && table->node[hash_value].busy==1){
            free(table->node[hash_value].info);
            free(table->node[hash_value].key);
            table->node[hash_value].busy=2;
            table->node[hash_value].info=NULL;
            table->node[hash_value].key=NULL;
            table->size--;
            return 0;
        }
    }
    return -1;
}

void kill(Table *table){
    for (int i; i<table->capacity; i++){
        if (table->node[i].key!=NULL) free(table->node[i].key);
    }
    free(table->node);
    free(table);
}
```
