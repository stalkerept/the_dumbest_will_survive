Дерево, где для любой вершины $x$ с ключом $k_{x}$:
- Все ключи в левом поддереве $\leq k_{x}$
- Все ключе в правом поддереве $\geq k_{x}$
Каждая вершина помечена отдельным ключом.
## Обход и поиск
- Центрированный обход выдаёт ключи в строго возрастающем порядке.
- Поиск по ключу: сравниваем с текущим узлом. Если <, то уходим влево, если >, то вправо, если == , то успех. Если дошли до NULL, то неудача.
- Поиск минимума: двигаемся по указателям left до NULL
- Поиск максимума: двигаемся по указателям right до NULL

### Оценка эффективности
Все операции поиска выполняются за O(h), где h - высота дерева.
- Худший случай: дерево вырождено в список, т.е. h = n.
- Средний/оптимальный случай: сбалансировано. Тогда $h = \approx \log_{2}{n}$

## Поиск предшествующего и последующего элементов.
**Предшествующий** - элемент с наибольшим ключом < заданного
**Последующий** - элемент с наименьшим ключом > заданного
Соответствуют соседям в последовательности **центрированного** обхода.

### Алгоритм поиска последующего (требует поле parent):
- Если есть правое поддерево: последующий = минимум правого поддерева (спуск влево до конца в правом поддереве)

- Если правого поддерева нет: поднимаемся вверх по родителям, пока текущий узел не станет левым ребёнком своего родителя. Родитель этого узла и будет последующим. Если же мы дошли до корня и не нашли, то значит последующего нет.
### Алгоритм поиска предшествующего (требует поле parent):
Симметричен:
- Если есть левое поддерево: предшествующий = максимум левого поддерева (спуск вправо до конца в левом поддереве)

- Если левого поддерева нет: поднимаемся вверх по родителям, пока текущий узел не станет правым ребёнком своего родителя. Родитель этого узла и будет предшествующим. Если же мы дошли до корня и не нашли, то значит последующего нет.

Оценка эффективности: В худшем случае O(n), в сбалансированном O($\log_{2}{n}$)

### Вставка нового элемента
- Если дерево пусто, то новый элемент становится корнем
- Иначе ищем позицию вставки, двигаясь от корня (как при коиске)
	- Запоминаем текущий узел и его родителя. Если значение ключа у текущего узла меньше, чем ключ у нового элемента, то уходим вправо. Иначе влево. 
	- Продолжаем до упора
- Когда упрёмся, присоединяем новый узел к последнему узлу как левое или правое поддерево

Оценка эффективности:
Поиск позиции O(h)
Присоединение: O(1)
Итог: O(h).


### Удаление элемента:
Найти реально удаляемый узел $y$:
Если у $x\leq 1$ ребёнка, то $y = x$.
Иначе $y =$ последующий элемент для $x$ (минимум правого поддерева). У $y$ гарантировано $\leq 1$ ребёнка.
Найти поддерево $p$ узла $y$ и родителя $par$.
Перенастроить связи:
- Если $p \neq$ NULL, то p -> parent = par
- Если $p ==$ NULL, то root = p (удалялся корень)
- Иначе par -> left/right = p
Копирование данных: Если $y \neq x$, скопировать key и info из $y$ в $x$.
Освободить память $y$.

Оценка эффективности:
Поиск $y$ и $par$: O(h)
Перенастройка + копирование: O(1)
Итог: O(h)

#include <stdio.h>

Stack* create_stack(void){

int insert(Tree *tree, KeyType key, InfoType info){
    if (tree==NULL || key==NULL) return -1;
    Node *node=(Node*)malloc(sizeof(Node));
    node->left=node->right=node->par=NULL;
    node->key=strdup(key);
    if (node->key==NULL) return -1;
    node->info=info;

        tree->root=node;
        return 0;

        par=cur;
        if (compare(key, cur->key)<0){
            cur=cur->left;
        }else{
            cur=cur->right;
        }
    }
    node->par=par;
    if (compare(key, par->key)<0){
        par->left=node;
    }else{
       par->right=node;
    }
    return 0;
}

int search(Tree *tree, KeyType key, Node **output){
    if (tree==NULL || key==NULL) return -1;
    if (tree->root==NULL){
        *output=NULL;
        return 0;
    }
    Node *cur=tree->root;
    Node *par=NULL;

        par=cur;
        if (compare(key, cur->key)==0){

            return 0;
        }
        if (compare(key, cur->key)<0){
            cur=cur->left;
        }else{
            cur=cur->right;
        }
    }
    return -1;
}

    if (tree==NULL || key==NULL || tree->root==NULL) return -1;
    Node *node;
    int i=search(tree, key, &node);
    if (i==-1) return -1;
    if (node==NULL) return -1;

        real_del=node;

    else child=real_del->right;
    if (child!=NULL) child->par=real_del->par;
    if (real_del->par==NULL) {
        tree->root=child;

        if (real_del==real_del->par->left) {
            real_del->par->left=child;
        } else {
            real_del->par->right=child;
        }
    }

        node->key=strdup(real_del->key);
        node->info=real_del->info;

    free(real_del);
    return 0;
}

int successor(Tree *tree, Node *node, Node **output){ //следующий
    if (tree==NULL || node==NULL || output==NULL) return -1;
    if (node->right!=NULL){
        Node *cur=node->right;
        while (cur->left!=NULL){
            cur=cur->left;
        }
        *output=cur;
        return 0;
    }
    Node* cur=node;
    Node* par=cur->par;
    while (par!=NULL && cur==par->right){
        cur=par;
        par=cur->par;
    }
    if (par==NULL) return -1;
    *output=par;
    return 0;
}

int predecessor(Tree *tree, Node *node, Node **output){ //предшествующий
    if (tree==NULL || node==NULL || output==NULL) return -1;
    if (node->left!=NULL){
        Node* cur=node->left;
        while (cur->right!=NULL){
            cur=cur->right;
        }
        *output=cur;
        return 0;
    }
    Node *cur=node;
    Node *par=cur->par;
    while (par!=NULL && cur==par->left){
        cur=par;
        par=cur->par;
    }
    if (par==NULL) return -1;
    *output=par;
    return 0;
}

    if (node == NULL) return;
    post_order_free(node->left);

    free(node->key);
    free(node);
}

void kill_tree(Tree* tree) {
    if (tree == NULL) return;
    post_order_free(tree->root);
